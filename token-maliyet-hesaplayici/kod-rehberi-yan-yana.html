55def select_models(pricing: dict, keys_csv: str | None) -> list[dict]:
56    """--models parametresine göre model listesini döndürür (boşsa hepsi)."""
57    models = pricing["models"]
58    if not keys_csv:
59        return models
60    by_key = {m["key"]: m for m in models}
61    selected = []
62    for key in (k.strip() for k in keys_csv.split(",") if k.strip()):
63        if key not in by_key:
64            valid = ", ".join(by_key)
65            raise ValueError(f"Bilinmeyen model anahtarı: '{key}'. Geçerli anahtarlar: {valid}")
66        selected.append(by_key[key])
67    return selected

127def render_bar(ratio: float, width: int = 20) -> str:
128    filled = min(width, round(min(ratio, 1.0) * width))
129    return "█" * filled + "░" * (width - filled)

144def fmt_usd(x: float) -> str:
145    if x == 0:
146        return "$0"
147    if x < 0:
148        return "-" + fmt_usd(-x)
149    # 1 doların altındaki tutarlarda ilk anlamlı basamaktan sonra bir basamak daha göster:
150    # 0,35 → $0,35  |  0,0035 → $0,0035  |  0,00012 → $0,00012
151    decimals = 2 if x >= 1 else min(8, -math.floor(math.log10(x)) + 1)
152    return "$" + _tr_number(f"{x:,.{decimals}f}")
