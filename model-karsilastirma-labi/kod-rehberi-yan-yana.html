100def stream_call(client, model_id: str, messages: list[dict], max_tokens: int, system: str | None = None,
101                extra: dict | None = None) -> dict:
102    """Streaming ile tek çağrı yapar; ilk token süresini, toplam süreyi ve usage değerlerini ölçer.
103
104    Süreler saniye cinsindendir. İlk token süresi, isteğin gönderilmesinden ilk metin
105    parçasının gelmesine kadar geçen süredir (Ders 1.4 §4.3). `extra`, modele özgü ek
106    istek ayarlarıdır (pricing.yaml > cagri_ayarlari); olduğu gibi isteğe eklenir.
107    """
108    kwargs = {"model": model_id, "max_tokens": max_tokens, "messages": messages}
109    if system:
110        kwargs["system"] = system
111    kwargs.update(extra or {})
112    start = time.perf_counter()
113    first = None
114    chunks = []
115    with client.messages.stream(**kwargs) as stream:
116        for text in stream.text_stream:
117            if first is None:
118                first = time.perf_counter()
119            chunks.append(text)
120        final = stream.get_final_message()
121    end = time.perf_counter()
122    return {
123        "text": "".join(chunks).strip(),
124        "ttft": (first or end) - start,
125        "total": end - start,
126        "input_tokens": final.usage.input_tokens,
127        "output_tokens": final.usage.output_tokens,
128        "stop_reason": final.stop_reason,
129    }
satır 100–129

stream_call: projenin kalbi

client.messages.stream yanıtı parça parça akıtır (Ders 1.4 §4.1). Çağrıdan hemen önce time.perf_counter() ile saat başlatılır; ilk metin parçası geldiğinde first işaretlenir, akış bitince end alınır. İlk token süresi first - start, toplam süre end - start'tır. get_final_message() akış bittikten sonra usage ve stop_reason alanlarını verir; akış sırasında bunlar henüz yoktur. kwargs sözlüğü, system yalnızca varsa eklensin diye kurulur; kwargs.update(extra or {}) modele özgü ek ayarları (pricing.yaml > cagri_ayarlari) isteğe katar ve **kwargs sözlüğü fonksiyona adlı parametreler olarak açar. Bu projede ek ayarlar düşünmeyi (thinking) en aza indirir: düşünme token'ları metin akışına girmez ama çıktı olarak faturalanır ve bütçeyi tüketir.

132def try_call(client, model: dict, messages: list[dict], max_tokens: int, system: str | None) -> tuple[dict | None, str | None]:

138    except anthropic.APIStatusError as e:
139        return None, f"HTTP {e.status_code}"
140    except anthropic.APIConnectionError:
141        return None, "bağlantı hatası"

try / except bloğu, içindeki kod hata verirse programın çökmesini önler. 175def check_output(text: str, checks: dict) -> tuple[float | None, list[str]]:
176    """Çıktıyı görevin kontrollerine göre puanlar: (0–100 puan, başarısız kontrollerin listesi).
177
178    Kontrol yoksa puan None döner. Her kontrol eşit ağırlıktadır.
179    """
180    if not checks:
181        return None, []
182    lowered = text.lower()
183    failed = []
184    for kind, rule in checks.items():
185        if kind == "icermeli":
186            missing = [s for s in rule if s.lower() not in lowered]
187            if missing:
188                failed.append(f"icermeli: {', '.join(missing)}")
189        elif kind == "icermeli_biri":
190            if not any(s.lower() in lowered for s in rule):
191                failed.append(f"icermeli_biri: {' | '.join(rule)}")
192        elif kind == "icermemeli":
193            present = [s for s in rule if s.lower() in lowered]
194            if present:
195                failed.append(f"icermemeli: {', '.join(present)}")
196        elif kind == "max_kelime":
197            words = len(text.split())
198            if words > rule:
199                failed.append(f"max_kelime: {words} > {rule}")
200        elif kind == "json":
201            parsed = _parse_json(text)
202            if not isinstance(parsed, dict):
203                failed.append("json: geçerli JSON nesnesi değil")
204            else:
205                missing = [k for k in rule if k not in parsed]
206                if missing:
207                    failed.append(f"json: eksik alan {', '.join(missing)}")
208    score = (len(checks) - len(failed)) / len(checks) * 100
209    return score, failed
satır 175–209

264def print_table(headers: list[str], rows: list[list[str]], right_align: set[int]) -> None:
265    widths = [max(len(str(cell)) for cell in column) for column in zip(headers, *rows)]

268        parts = []
269        for i, cell in enumerate(cells):

378def summarize(records: list[dict]) -> list[dict]:
379    """Çalıştırma kayıtlarını model başına özetler; etiketsiz puanlar varsa ortalamasını ekler."""
380    runs = [r for r in records if r.get("tur") == "calistirma"]
381    ratings = [r for r in records if r.get("tur") == "puan"]
382    summaries = []
383    for key in dict.fromkeys(r["model"] for r in runs):
384        mine = [r for r in runs if r["model"] == key]
385        scored = [r["puan"] for r in mine if r["puan"] is not None]
386        rated = [r["puan"] for r in ratings if r["model"] == key]
387        totals = [r["toplam_sn"] for r in mine]

389        summaries.append({
390            "model": key, "model_adi": mine[0]["model_adi"], "calistirma": len(mine),
391            "tekrar": max(r.get("tekrar", 1) for r in mine),
392            "ilk_token_sn": mean([r["ilk_token_sn"] for r in mine]),
393            "toplam_sn": mean(totals), "toplam_min_sn": min(totals), "toplam_max_sn": max(totals),
394            "token_sn": mean(speeds) if speeds else None,
395            "cikti_token": mean([r["cikti_token"] for r in mine]),
396            "maliyet": mean([r["maliyet"] for r in mine]),
397            "kesilen": sum(1 for r in mine if r["kesildi"]),

513def parse_scores(scores_csv: str, labels: list[str]) -> dict[str, int]:
514    """'A=4,B=3' biçimindeki metni {etiket: puan} sözlüğüne çevirir; eksik ya da geçersiz girdiyi reddeder."""
515    scores = {}
516    for part in (p.strip() for p in scores_csv.split(",") if p.strip()):
517        label, _, value = part.partition("=")
518        label = label.strip().upper()
519        if label not in labels or not value.strip().isdigit() or not 1 <= int(value) <= 5:
520            raise ValueError(f"Geçersiz puan girdisi: '{part}'. Biçim: A=4 (1–5 arası).")
521        scores[label] = int(value)
522    missing = [label for label in labels if label not in scores]
523    if missing:
524        raise ValueError(f"Eksik puan: {', '.join(missing)}. Her çıktıya puan ver.")
525    return scores

Programın başladığı yer

Dosya doğrudan çalıştırıldığında __name__ değeri "__main__" olur ve main() çağrılır. Testler dosyayı import ettiğinde bu koşul sağlanmaz ve hiçbir şey kendiliğinden çalışmaz. sys.exit dönen sayıyı terminale çıkış kodu olarak iletir.

Kodu ilk kez okuyorsan en alttaki "Programın başladığı yer" bölümünden başlayıp yukarı doğru, çağrılan fonksiyonları takip ederek ilerlemeyi de deneyebilirsin. Fiyat ve model değerleri pricing.yaml dosyasından okunur.
