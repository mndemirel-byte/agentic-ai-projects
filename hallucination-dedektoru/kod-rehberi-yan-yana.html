107def load_pricing(path: Path) -> dict:
108    """pricing.yaml dosyasını okur; zorunlu alanları ve rol atamalarını doğrular."""
109    with open(path, encoding="utf-8") as f:
110        data = yaml.safe_load(f)
111    for model in data["models"]:
112        missing = [field for field in REQUIRED_MODEL_FIELDS if field not in model]

174def call_model(client, model: dict, prompt: str, max_tokens: int, system: str | None = None) -> dict:
175    """Tek bir non-streaming çağrı yapar; metni, token sayılarını ve durma nedenini döndürür.
176
177    Modelin `cagri_ayarlari` sözlüğü (pricing.yaml) isteğe olduğu gibi eklenir; kod ayarların
178    içeriğini bilmez. Yanıt bütün olarak işleneceği için streaming gerekmez (Ders 1.4 §4.4).
179    """
180    kwargs = {"model": model["id"], "max_tokens": max_tokens, "messages": [{"role": "user", "content": prompt}]}
181    if system:
182        kwargs["system"] = system
183    kwargs.update(model.get("cagri_ayarlari") or {})
184    message = client.messages.create(**kwargs)
185    text = "".join(block.text for block in message.content if getattr(block, "type", "") == "text")
186    return {
187        "text": text.strip(),
188        "input_tokens": message.usage.input_tokens,
189        "output_tokens": message.usage.output_tokens,
190        "stop_reason": message.stop_reason,
191    }
satır 170–191

call_model: API ile konuşan tek fonksiyon

kwargs sözlüğü isteğin parametrelerini toplar; system yalnızca varsa eklenir; kwargs.update(model.get("cagri_ayarlari") or {}) modele özgü ek ayarları (pricing.yaml) isteğe katar. Kod ayarların içeriğini bilmez; bir model yeni bir parametre isterse yalnızca YAML değişir. client.messages.create(**kwargs) sözlüğü adlı parametreler olarak açar ve yanıtı bütün olarak alır; yanıt bütün işleneceği için streaming gerekmez (Ders 1.4 §4.4). content bir blok dizisidir; yalnızca text tipindeki bloklar birleştirilir.

211def verify_claims(client, model: dict, source: str, claims: list[str]) -> tuple[list[dict], dict]:
212    """Zincirin 2. aşamasının LLM yarısı: her iddiayı kaynağa karşı değerlendirir."""
213    numbered = "\n".join(f"{i}. {claim}" for i, claim in enumerate(claims, start=1))
214    prompt = VERIFY_PROMPT.replace("{source}", source).replace("{claims}", numbered)
215    result = call_model(client, model, prompt, MAX_TOKENS["verify"])
216    parsed = _parse_json(result["text"])
217    rows = parsed.get("results") if isinstance(parsed, dict) else None
218    if not isinstance(rows, list) or len(rows) != len(claims):
219        raise ValueError(f"Doğrulama: model {len(claims)} iddia için {len(rows) if isinstance(rows, list) else 0} "
220                         "sonuç döndürdü; geçerli JSON bekleniyordu.")
221    verdicts = []
222    for claim, row in zip(claims, rows):
223        verdict = str(row.get("verdict", "")).strip().lower()
224        if verdict not in ("supported", "unsupported", "contradicted"):
225            raise ValueError(f"Doğrulama: bilinmeyen karar '{verdict}' (iddia: {claim[:60]})")
226        verdicts.append({"iddia": claim, "karar": verdict, "kanit": str(row.get("evidence") or "").strip(),
227                         "not": str(row.get("note") or "").strip()})
228    return verdicts, result

240def _parse_json(text: str):
241    """Model çıktısındaki JSON'u çözer; kod bloğu çitlerini (```json ... ```) ve JSON'un çevresindeki metni atar.
242
243    Çözülemezse None döner. Önce metnin tamamı, sonra ilk '{' ile son '}' arasındaki parça denenir;
244    böylece "İşte iddialar:" gibi bir giriş cümlesi ayrıştırmayı bozmaz.
245    """
246    cleaned = text.strip()
247    if cleaned.startswith("```"):
248        cleaned = re.sub(r"^```[a-zA-Z]*\s*|\s*```$", "", cleaned)
249    candidates = [cleaned]
250    if "{" in cleaned and "}" in cleaned:
251        candidates.append(cleaned[cleaned.find("{"): cleaned.rfind("}") + 1])
252    for candidate in candidates:
253        try:
254            return json.loads(candidate)
